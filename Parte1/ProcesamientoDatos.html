#3. Limpieza

df_limpia <- df_sucia %>%
  # 1. Eliminar filas idénticas
  distinct() %>%
  
  # 2. Filtrar filas sin título o sin argumento
  filter(!is.na(Title), Title != "", !is.na(Plot), Plot != "") %>%
  
  # 3. Limpiar espacios y pasar a minúsculas (Sintaxis corregida)
  mutate(
    Title = tolower(trimws(Title)),
    Plot = tolower(trimws(Plot)),
    Director = tolower(trimws(Director)),
    Genre = tolower(trimws(Genre))
  ) %>%
  
  # 4. Remover acentos y caracteres raros
  mutate(
    Title = iconv(Title, from = "UTF-8", to = "ASCII//TRANSLIT"),
    Plot = iconv(Plot, from = "UTF-8", to = "ASCII//TRANSLIT")
  ) %>%
  
  # 5. Evitar duplicados reales (Mismo título Y mismo año)
  distinct(Title, `Release Year`, .keep_all = TRUE)

# Eliminar saltos de línea (\n y \r) dentro de los textos antes de guardar
df_para_cpp <- df_limpia %>%
  mutate(
    Plot = gsub("[\r\n]", " ", Plot),
    Director = gsub("[\r\n]", " ", Director)
  )

# Guardar usando la versión moderna de tidyverse
write_csv(df_para_cpp, "movies_clean.csv", na = "")
